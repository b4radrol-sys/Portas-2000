void bekerauto()
{
    Console.WriteLine("-----------------");

    int sorszam = File.ReadAllLines("autoadatok.txt").Length;

    Console.Write("Márka: ");
    string marka = Console.ReadLine();

    while (marka.Any(char.IsDigit) || string.IsNullOrWhiteSpace(marka))
    {
        Console.WriteLine("---Helytelen megadás---");
        Console.Write("Márka: ");
        marka = Console.ReadLine();
    }



    Console.Write("Tipus: ");
    string tipus = Console.ReadLine();

    while (string.IsNullOrWhiteSpace(tipus))
    {
        Console.WriteLine("---Helytelen megadás---");
        Console.Write("Tipus: ");
        tipus = Console.ReadLine();
    }


    Console.Write("Rendszáma: ");
    string rendszam = Console.ReadLine();

    while (string.IsNullOrWhiteSpace(rendszam))
    {
        Console.WriteLine("---Helytelen megadás---");
        Console.Write("Rendszám: ");
        rendszam = Console.ReadLine();
    }



    Console.Write("Szín: ");
    string szin = (Console.ReadLine());

    while (szin.Any(char.IsDigit) || string.IsNullOrWhiteSpace(szin))
    {
        Console.WriteLine("---Helytelen megadás---");
        Console.Write("Szín: ");
        szin = Console.ReadLine();
    }



    List<string> autolista = new() { sorszam + "," + marka + "," + tipus + "," + rendszam + "," + szin };
    File.AppendAllLines("autoadatok.txt", autolista);
    Console.WriteLine("---Sikeresen hozzáadva---");
    Console.WriteLine("-----------------");
}

void modosit()
{
    Console.Write("Mit akar szerkeszteni? (1: auto, 2: szemely): ");
    int modositacsocska = int.Parse(Console.ReadLine());

    string utvonalocska = "";
    if (modositacsocska == 1)
    {
        utvonalocska = "autoadatok.txt";
    }
    else if (modositacsocska == 2)
    {
        utvonalocska = "emberadatok.txt";
    }
    else
    {
        Console.WriteLine("Érvénytelen választás!");
        return;
    }

    if (!File.Exists(utvonalocska))
    {
        Console.WriteLine("Fájl nem található!");
        return;
    }


    string[] sorok = File.ReadAllLines(utvonalocska);


    Console.WriteLine("\n--- Jelenlegi adatok ---");
    foreach (string sor in sorok)
    {
        Console.WriteLine(sor);
    }

    Console.Write("\nAdja meg a módosítani kívánt ID-t: ");
    string keresettId = Console.ReadLine();

    bool talalt = false;


    for (int i = 1; i < sorok.Length; i++)
    {
        string[] reszek = sorok[i].Split(',');


        if (reszek[0].Trim() == keresettId)
        {
            talalt = true;
            Console.WriteLine($"\nMódosítás erre a rekordra: {sorok[i]}");

            if (modositacsocska == 1)
            {
                Console.Write("Új Márka: ");
                string ujMarka = Console.ReadLine();

                Console.Write("Új Típus: ");
                string ujTipus = Console.ReadLine();

                Console.Write("Új Rendszám: ");
                string ujRendszam = Console.ReadLine();

                Console.Write("Új Szín: ");
                string ujSzin = Console.ReadLine();


                sorok[i] = $"{keresettId},{ujMarka},{ujTipus},{ujRendszam},{ujSzin}";
            }
            else if (modositacsocska == 2)
            {
                Console.Write("Új Név: ");
                string ujNev = Console.ReadLine();

                Console.Write("Új Neme: ");
                string ujNem = Console.ReadLine();

                Console.Write("Új Kor: ");
                string ujKor = Console.ReadLine();

                Console.Write("Új Hajszín: ");
                string ujHajszin = Console.ReadLine();

                Console.Write("Új Igazolványszám: ");
                string ujIgazolvany = Console.ReadLine();


                sorok[i] = $"{keresettId},{ujNev},{ujNem},{ujKor},{ujHajszin},{ujIgazolvany}";
            }

            File.WriteAllLines(utvonalocska, sorok);
            Console.WriteLine("--- Sikeresen módosítva! ---");
            break;
        }
    }

    if (!talalt)
    {
        Console.WriteLine("Nincs ilyen ID-jű rekord!");
    }
}

void beker()
    {
        Console.Write("Ember vagy Autó érkezett(1. ember 2.autó)? ");
        byte bekerki = 0;

        while (bekerki != 1 || bekerki != 2)
        {
            bekerki = byte.Parse(Console.ReadLine());
            if (bekerki == 1)
            {
                bekerember();
            }
            else if (bekerki == 2)
            {
                bekerauto();
            }
            else
            {
                Console.WriteLine("Kérem 1 vagy 2-t írjon be");
            }
        }
    }
